if(ESP_PLATFORM)
    idf_component_register(SRCS "src/tone_noise_generator.cc" INCLUDE_DIRS "include")
    return()
endif()
cmake_minimum_required(VERSION 3.20)
project(esp32_local_sounds LANGUAGES CXX)
add_library(tone_noise_generator src/tone_noise_generator.cc)
target_include_directories(tone_noise_generator PUBLIC include)
target_compile_features(tone_noise_generator PUBLIC cxx_std_17)
add_executable(render_pcm examples/render_pcm.cc)
target_link_libraries(render_pcm PRIVATE tone_noise_generator)
include(CTest)
if(BUILD_TESTING)
    foreach(name generator synth volume ocean saw)
        add_executable(${name}_tests tests/${name}_tests.cc)
        target_link_libraries(${name}_tests PRIVATE tone_noise_generator)
        add_test(NAME ${name}_tests COMMAND ${name}_tests)
    endforeach()
    foreach(name adapter tools resume)
        add_executable(${name}_tests tests/${name}_tests.cc
            integrations/xiaozhi/generated_audio_adapter.cc
            integrations/xiaozhi/local_audio_adapter.cc)
        target_include_directories(${name}_tests PRIVATE tests/support)
        target_link_libraries(${name}_tests PRIVATE tone_noise_generator)
        if(name STREQUAL "tools")
            target_sources(${name}_tests PRIVATE integrations/xiaozhi/generated_audio_tools.cc)
            set_property(TARGET ${name}_tests PROPERTY CXX_STANDARD 23)
        endif()
        add_test(NAME ${name}_tests COMMAND ${name}_tests)
    endforeach()
endif()
